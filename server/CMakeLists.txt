set(SERVER_APP_NAME Server)
set(CORE_LIB_NAME Core)
set(LIBSODIUM_VENDOR_NAME sodium)
set(STDUUID_VENDOR_NAME stduuid)
  

file(GLOB_RECURSE AUTH_SOURCE_FILES "src/auth/*.cpp")
file(GLOB_RECURSE CONVERSATIONS_SOURCE_FILES "src/conversations/*.cpp")
file(GLOB_RECURSE ENCRYPTION_SOURCE_FILES "src/encryption/*.cpp")
file(GLOB_RECURSE EXCEPTIONS_SOURCE_FILES "src/exceptions/*.cpp")
file(GLOB_RECURSE MESSAGES_SOURCE_FILES "src/messages/*.cpp")
file(GLOB_RECURSE MIDDLEWARES_SOURCE_FILES "src/middlewares/*.cpp")
file(GLOB_RECURSE SESSIONS_SOURCE_FILES "src/sessions/*.cpp")
file(GLOB_RECURSE USERS_SOURCE_FILES "src/users/*.cpp")
file(GLOB_RECURSE UUID_SOURCE_FILES "src/uuid/*.cpp")

add_executable(${SERVER_APP_NAME} ${AUTH_SOURCE_FILES} ${CONVERSATIONS_SOURCE_FILES} ${ENCRYPTION_SOURCE_FILES} ${EXCEPTIONS_SOURCE_FILES} ${MESSAGES_SOURCE_FILES} ${MIDDLEWARES_SOURCE_FILES} ${SESSIONS_SOURCE_FILES} ${USERS_SOURCE_FILES} ${UUID_SOURCE_FILES} src/main.cpp)
target_link_libraries(${SERVER_APP_NAME} PRIVATE ${CORE_LIB_NAME} ${LIBSODIUM_VENDOR_NAME} ${STDUUID_VENDOR_NAME})
target_include_directories(${SERVER_APP_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
